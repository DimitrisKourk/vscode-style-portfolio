import { createPortfolio } from "./portfolio.js";

const name = "Dimitrios Kourkoutas";

async function welcome(name) {
  const information = await fetch(`https://api.portfolio.dev/v1/users/${name}`)    
  .then((response) => response.json())
    .then((data) => ({
      name: data.name,           // "Dimitrios Kourkoutas"
      title: data.title,         // "Full Stack Software Engineer"
      mail: data.mail,           // "dimitriskourkoutas@yahoo.gr"
      linkedin: data.linkedin,   // "linkedin.com/in/dimitriskourkoutas
      github: data.github,       // "github.com/DimitrisKourk"
      bio: data.bio,             // "Turning ideas into production-ready software."
      projects: data.projects,   // ["Real-estate SaaS", "Photo Michalopoulos", "Chef for Crews"]
    }));

  try {
    await createPortfolio(information);
    console.log(`Welcome ${name} to your portfolio website!`);
    return;
  } catch (error) {
    console.error(`Could not create the portfolio: ${error.message}`);
    return;
  }
}

welcome(name);
